<script>
  // === MOBILE MENU LOGIC ===
  const menuBtn = document.getElementById("menuToggle");
  const mobileNav = document.getElementById("mobileNav");

  menuBtn.addEventListener("click", () => {
    menuBtn.classList.toggle("active"); // Triggers the Cross animation
    mobileNav.classList.toggle("active"); // Slides down the menu
  });

  // === FILTER ANIMATION LOGIC ===
  function filterProducts(category) {
    // 1. Update Buttons
    document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
    event.target.classList.add('active');

    // 2. Filter Cards with Delay
    const cards = document.querySelectorAll('.product-card');
    let delayCounter = 0;

    cards.forEach(card => {
      card.style.animation = 'none'; // Reset animation
      card.offsetHeight; /* Trigger reflow */
      
      if (category === 'all' || card.classList.contains(category)) {
        card.classList.remove('hide');
        card.style.animation = `popIn 0.5s cubic-bezier(0.25, 0.46, 0.45, 0.94) forwards`;
        card.style.animationDelay = `${delayCounter * 0.1}s`;
        delayCounter++;
      } else {
        card.classList.add('hide');
      }
    });
  }

  // === MODAL LOGIC ===
  const modal = document.getElementById("productModal");
  const mTitle = document.getElementById("modalTitle");
  const mDesc = document.getElementById("modalDesc");
  const mPack = document.getElementById("modalPack");

  function openModal(name, desc, form) {
    mTitle.textContent = name;
    mDesc.textContent = desc;
    mPack.textContent = form === 'Liquid' ? '30kg / 50kg Carboys' : '20kg / 25kg Cartons';
    modal.classList.add("active");
  }

  function closeModal() {
    modal.classList.remove("active");
  }

  // Close on outside click
  window.onclick = function(event) {
    if (event.target == modal) {
      closeModal();
    }
  }
</script>
